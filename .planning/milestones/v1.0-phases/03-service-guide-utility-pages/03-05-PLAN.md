  const grouped: Record<string, ServiceContent[]> = {}
  for (const service of Object.values(serviceContentMap)) {
    if (!grouped[service.category]) grouped[service.category] = []
    grouped[service.category].push(service)
  }
  return grouped
}

**app/services/[serviceSlug]/page.tsx** -- Dynamic route template:

Follow Phase 2's `app/[locationSlug]/page.tsx` pattern exactly for the Next.js 16 async params pattern.

```typescript
import { getServiceContent, getAllServiceSlugs } from '@/data/services/content'
import { notFound } from 'next/navigation'
import type { Metadata } from 'next'
// Import all section components and schema builders

export const dynamicParams = false

export function generateStaticParams() {
  return getAllServiceSlugs().map((slug) => ({ serviceSlug: slug }))
}

export async function generateMetadata({
  params,
}: {
  params: Promise<{ serviceSlug: string }>
}): Promise<Metadata> {
  const { serviceSlug } = await params
  const content = getServiceContent(serviceSlug)
  if (!content) return {}

  return {
    title: content.seoTitle,
    description: content.seoDescription,
    alternates: { canonical: `/services/${serviceSlug}` },
    openGraph: {
      title: content.seoTitle,
      description: content.seoDescription,
      url: `/services/${serviceSlug}`,
    },
  }
}

export default async function ServicePage({
  params,
}: {
  params: Promise<{ serviceSlug: string }>
}) {
  const { serviceSlug } = await params
  const content = getServiceContent(serviceSlug)
  if (!content) notFound()

  // Build schemas
  const serviceSchema = buildServiceSchema(content.name, content.seoDescription, content.slug)
  const faqSchema = buildFaqSchema(content.faqItems)
  const breadcrumbItems = [
    { name: 'Home', url: '/' },
    { name: 'Services', url: '/services' },
    { name: content.name, url: `/services/${content.slug}` },
  ]

  // Resolve related services from data/services.ts lightweight index
  const relatedServices = content.relatedServiceSlugs
    .map((slug) => services.find((s) => s.slug === slug))
    .filter(Boolean)

  // Render 13 sections in D-01 order (see UI-SPEC page section order)
  return (
    <>
      <JsonLd data={serviceSchema} />
      <JsonLd data={faqSchema} />
      <Breadcrumbs items={breadcrumbItems} />
      <Hero headline={content.heroHeadline} subheadline={content.heroSubheadline} />
      <ServiceOverview overviewHtml={content.overviewHtml} serviceName={content.name} />
      <MidPageCTA />  {/* After overview */}
      <ProcessSteps steps={content.processSteps} serviceName={content.name} />
      <BenefitsGrid benefits={content.benefits} serviceName={content.name} />
      <MidPageCTA />  {/* After benefits */}
      <RelatedLocations serviceName={content.name} />
      <RelatedServices services={relatedServices} />
      <ServiceFAQ faqItems={content.faqItems} serviceName={content.name} />
      <Testimonials />
      <EmergencyCTA />
      <ContactForm />
    </>
  )
}
```

Note: Import `services` from `@/data/services` for resolving `relatedServiceSlugs` to full service objects for the RelatedServices component.
